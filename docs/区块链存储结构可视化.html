// 链码中的序列化过程
accessLog := &model.PrescriptionAccessLog{...}
accessLogBytes, err := json.Marshal(accessLog) // 转为JSON字节数组

// 创建复合主键
key := stub.CreateCompositeKey("access-log-key", []string{logID})
// 结果: "access-log-key\x007556dec5-c99f-4ddb-88c4-a8dba40ccbfb\x00"

// 写入区块链账本
stub.PutState(key, accessLogBytes) // 永久存储，不可篡改

// 获取交易ID
txID := stub.GetTxID() // efc0b928b2e9f817a2a33b1fa86697a3...
