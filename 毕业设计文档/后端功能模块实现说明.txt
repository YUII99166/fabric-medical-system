基于区块链的医疗信息管理系统 - 后端功能模块实现说明

7.2.1 用户登录模块

当用户输入账号密码登录时，LoginWithPassword方法会接收用户名、密码作为参数。首先，它通过ShouldBind方法解析请求体参数，并验证用户名和密码是否为空。接着，调用db.LoginUser方法从数据库中查询用户信息，并检查用户是否存在以及密码是否正确。如果用户验证失败，返回一个错误信息；如果验证成功，则返回包含用户ID、用户名、账户名称、角色等信息的成功响应。核心代码如下：

func LoginWithPassword(c *gin.Context) {
	appG := app.Gin{C: c}
	body := new(model.LoginBody)

	// 解析Body参数
	if err := c.ShouldBind(body); err != nil {
		appG.Response(http.StatusBadRequest, "失败", 
			fmt.Sprintf("参数解析错误: %s", err.Error()))
		return
	}
	
	// 验证用户名和密码不能为空
	if body.Username == "" || body.Password == "" {
		appG.Response(http.StatusBadRequest, "失败", "用户名或密码不能为空")
		return
	}

	// 从数据库验证用户
	user, err := db.LoginUser(body.Username, body.Password)
	if err != nil {
		if err.Error() == "用户名或密码错误" {
			appG.Response(http.StatusUnauthorized, "失败", 
				"用户名或密码错误，请检查后重试")
		} else {
			appG.Response(http.StatusInternalServerError, "失败", 
				fmt.Sprintf("登录失败: %v", err))
		}
		return
	}

	// 构建返回数据
	data := map[string]interface{}{
		"id":           user.ID,
		"username":     user.Username,
		"account_name": user.AccountName,
		"role":         user.Role,
		"created_at":   user.CreatedAt,
		"updated_at":   user.UpdatedAt,
	}

	appG.Response(http.StatusOK, "成功", data)
}


7.2.2 用户注册模块

当用户提交注册信息时，Register方法首先解析请求体参数，验证必填字段（用户名、密码、账户名称、角色）是否完整。接着，检查用户名是否已存在，如果存在则返回错误。然后，调用区块链服务创建用户账户，生成区块链ID。最后，将用户信息保存到数据库中，并返回包含用户ID和区块链ID的成功响应。核心代码如下：

func Register(c *gin.Context) {
	appG := app.Gin{C: c}
	body := new(model.RegisterBody)

	// 解析请求参数
	if err := c.ShouldBind(body); err != nil {
		appG.Response(http.StatusBadRequest, "失败", 
			fmt.Sprintf("参数解析错误: %s", err.Error()))
		return
	}

	// 验证必填字段
	if body.Username == "" || body.Password == "" || 
	   body.AccountName == "" || body.Role == "" {
		appG.Response(http.StatusBadRequest, "失败", "必填字段不能为空")
		return
	}

	// 检查用户名是否已存在
	exists, err := db.CheckUsernameExists(body.Username)
	if err != nil {
		appG.Response(http.StatusInternalServerError, "失败", 
			fmt.Sprintf("检查用户名失败: %v", err))
		return
	}
	if exists {
		appG.Response(http.StatusBadRequest, "失败", "用户名已存在")
		return
	}

	// 调用区块链服务创建账户
	blockchainID, txID, err := blockchain.CreateAccount(body)
	if err != nil {
		appG.Response(http.StatusInternalServerError, "失败", 
			fmt.Sprintf("区块链账户创建失败: %v", err))
		return
	}

	// 保存到数据库
	userID, err := db.CreateUser(body, blockchainID, txID)
	if err != nil {
		appG.Response(http.StatusInternalServerError, "失败", 
			fmt.Sprintf("数据库保存失败: %v", err))
		return
	}

	// 返回成功响应
	data := map[string]interface{}{
		"id":            userID,
		"blockchain_id": blockchainID,
		"tx_id":         txID,
	}
	appG.Response(http.StatusOK, "成功", data)
}


7.2.3 病历创建模块

当医生创建病历时，CreatePrescription方法首先解析请求体参数，验证必填字段（患者ID、医生ID、诊断等）是否完整。接着，调用区块链服务将病历数据上链，生成唯一的病历ID和交易ID。然后，将病历信息保存到数据库中进行缓存。最后，记录审计日志并返回包含病历ID和区块链交易ID的成功响应。核心代码如下：

func CreatePrescription(c *gin.Context) {

			fmt.Sprintf("参数解析错误: %s", err.Error()))

	// 验证必填字段

7.2.4 病历查询模块

当用户查询病历列表时，QueryPrescriptionList方法首先解析查询参数（如患者ID、医生ID、时间范围等）。接着，根据用户角色判断查询权限：患者只能查询自己的病历，医生可以查询自己创建的病历或已授权的病历。然后，调用数据库服务进行分页查询，获取符合条件的病历列表。最后，返回包含病历列表和分页信息的成功响应。核心代码如下：

func QueryPrescriptionList(c *gin.Context) {

	body := new(model.PrescriptionQueryRequestBody)

	// 解析查询参数

		appG.Response(http.StatusBadRequest, "失败", 
			fmt.Sprintf("参数解析错误: %s", err.Error()))
		return

	// 获取当前用户信息
	userID := c.GetString("user_id")
	userRole := c.GetString("user_role")

		"size":  body.Size,

	appG.Response(http.StatusOK, "成功", data)
}


7.2.5 授权请求模块

当医生申请查看其他医院患者的病历时，RequestAccess方法首先解析请求参数（病历ID、医生ID、申请理由等）。接着，验证必填字段是否完整，确保病历ID、医生ID和申请理由都不为空。然后，调用区块链智能合约requestAccess方法创建授权请求，该方法采用重试机制处理区块链网络的临时故障。最后，将区块链返回的授权请求数据解析并返回给前端。患者可以在系统中查看并审批该请求。核心代码如下：

func RequestAccess(c *gin.Context) {
	appG := app.Gin{C: c}
	body := new(model.AccessRequestBody)

	// 解析并验证请求参数
	if err := c.ShouldBind(body); err != nil {
		appG.Response(http.StatusBadRequest, "失败", 
			fmt.Sprintf("参数解析错误: %s", err.Error()))
		return
	}
	if body.PrescriptionID == "" || body.DoctorID == "" || body.Reason == "" {
		appG.Response(http.StatusBadRequest, "失败", "参数存在空值")
		return
	}

	// 构建区块链调用参数并执行
	var bodyBytes [][]byte
	bodyBytes = append(bodyBytes, []byte(body.PrescriptionID))
	bodyBytes = append(bodyBytes, []byte(body.DoctorID))
	bodyBytes = append(bodyBytes, []byte(body.Reason))

	resp, err := bc.ChannelExecute("requestAccess", bodyBytes)
	if err != nil {
		appG.Response(http.StatusInternalServerError, "失败", 
			fmt.Sprintf("区块链调用失败: %v", err))
		return
	}
	
	// 解析并返回结果
	var data map[string]interface{}
	json.Unmarshal(resp.Payload, &data)
	appG.Response(http.StatusOK, "成功", data)
}

7.2.7 病历访问日志模块

当用户查看病历时，系统自动调用RecordPrescriptionAccess方法记录访问日志。该方法首先获取当前用户信息和访问的病历ID，然后构建访问日志数据（包括访问者ID、访问类型、IP地址等）。接着，调用区块链服务将访问日志上链，确保日志不可篡改。最后，将访问日志保存到数据库，供患者查询隐私溯源信息。核心代码如下：

func RecordPrescriptionAccess(c *gin.Context) {
	appG := app.Gin{C: c}

	// 获取访问信息
	prescriptionID := c.Param("prescription_id")
	accessorID := c.GetString("user_id")
	accessorRole := c.GetString("user_role")
	ipAddress := c.ClientIP()

	// 构建访问日志数据
	logData := map[string]interface{}{
		"prescription_id": prescriptionID,
		"accessor_id":     accessorID,
		"accessor_role":   accessorRole,
		"access_type":     "view",
		"ip_address":      ipAddress,
		"accessed_at":     time.Now(),
	}

	// 调用区块链服务记录访问日志
	logID, txID, err := blockchain.RecordAccessLog(logData)
	if err != nil {
		// 记录日志失败不影响主流程，只记录错误
		log.Printf("记录访问日志失败: %v", err)
		return
	}

	// 保存到数据库
	err = db.SaveAccessLog(logData, logID, txID)
	if err != nil {
		log.Printf("保存访问日志失败: %v", err)
	}
}
